-- ==============================================================================
-- 1. BASE INDEPENDENT TABLES & NORMALIZED ROOM CHAIN
-- ==============================================================================

-- From: ROOMS_B (Capacity, Availability)
CREATE TABLE ROOMS_B (
    Capacity INT PRIMARY KEY,
    Availability VARCHAR(50)
);

-- From: ROOMS (Room_no, Capacity, Type)
-- Constraint applied: Links to ROOMS_B
CREATE TABLE ROOMS (
    Room_no INT PRIMARY KEY,
    Capacity INT,
    Type VARCHAR(50),
    CONSTRAINT fk_rooms_capacity FOREIGN KEY (Capacity) REFERENCES ROOMS_B(Capacity)
);

-- From: EMPLOYEE (Emp_id, F_name, L_name, Address, Contact_no, Age)
CREATE TABLE EMPLOYEE (
    Emp_id INT PRIMARY KEY,
    F_name VARCHAR(50),
    L_name VARCHAR(50),
    Gender VARCHAR(10),
    Address VARCHAR(255),
    Contact_no VARCHAR(15),
    Age INT
);

-- ==============================================================================
-- 2. NORMALIZED PATIENT CHAIN (Satisfies the "Assigned" Relationship to Rooms)
-- ==============================================================================

-- From: PATIENT_C (Address, room_no)
CREATE TABLE PATIENT_C (
    Address VARCHAR(255) PRIMARY KEY,
    Room_no INT,
    CONSTRAINT fk_patientc_room FOREIGN KEY (Room_no) REFERENCES ROOMS(Room_no)
);

-- From: PATIENT_B (Phone, Address)
CREATE TABLE PATIENT_B (
    Phone VARCHAR(15) PRIMARY KEY,
    Address VARCHAR(255),
    CONSTRAINT fk_patientb_address FOREIGN KEY (Address) REFERENCES PATIENT_C(Address)
);

-- From: PATIENT (Patient_id, F_name, L_name, Gender, Phone, in_date, out_date)
-- EER Constraint: Base entity for Patient
CREATE TABLE PATIENT (
    Patient_id INT PRIMARY KEY,
    F_name VARCHAR(50),
    L_name VARCHAR(50),
    Gender VARCHAR(10),
    Phone VARCHAR(15),
    In_date DATE,
    Out_date DATE,
    CONSTRAINT fk_patient_phone FOREIGN KEY (Phone) REFERENCES PATIENT_B(Phone)
);

-- ==============================================================================
-- 3. EMPLOYEE SUBCLASSES (Satisfying EER Inheritance constraints)
-- ==============================================================================

-- From: DOCTOR (Emp_id, Specialization, Designation, Supervisor_id)
-- EER Constraint: Subclass of Employee, Recursive Relationship (Supervision)
CREATE TABLE DOCTOR (
    Emp_id INT PRIMARY KEY,
    Specialization VARCHAR(100),
    Designation VARCHAR(100),
    Supervisor_id INT,
    CONSTRAINT fk_doctor_emp FOREIGN KEY (Emp_id) REFERENCES EMPLOYEE(Emp_id),
    CONSTRAINT fk_doctor_supervisor FOREIGN KEY (Supervisor_id) REFERENCES DOCTOR(Emp_id)
);

-- From: NURSE (Emp_id, Shift_type)
-- EER Constraint: Subclass of Employee
CREATE TABLE NURSE (
    Emp_id INT PRIMARY KEY,
    Shift_type VARCHAR(50),
    CONSTRAINT fk_nurse_emp FOREIGN KEY (Emp_id) REFERENCES EMPLOYEE(Emp_id)
);

-- From: PHARMACIST (Emp_id, Clearance_level)
-- EER Constraint: Subclass of Employee
CREATE TABLE PHARMACIST (
    Emp_id INT PRIMARY KEY,
    Clearance_level VARCHAR(50),
    CONSTRAINT fk_pharmacist_emp FOREIGN KEY (Emp_id) REFERENCES EMPLOYEE(Emp_id)
);

-- ==============================================================================
-- 4. WEAK ENTITIES (Satisfying EER Identifying Relationships to Patient)
-- ==============================================================================

-- From: PERSONS (Patient_id, Name, sex, Age, Relationship)
-- EER Constraint: Weak entity on Patient ('Dependent on')
CREATE TABLE PERSONS (
    Patient_id INT,
    Name VARCHAR(100),
    Sex VARCHAR(10),
    Age INT,
    Relationship VARCHAR(50),
    PRIMARY KEY (Patient_id, Name),
    CONSTRAINT fk_persons_patient FOREIGN KEY (Patient_id) REFERENCES PATIENT(Patient_id)
);

-- From: TEST_REPORT (R_id, Patient_id, Test_type, Result)
-- EER Constraint: Weak entity on Patient ('Has')
CREATE TABLE TEST_REPORT (
    R_id INT,
    Patient_id INT,
    Test_type VARCHAR(100),
    Result VARCHAR(255),
    PRIMARY KEY (R_id, Patient_id),
    CONSTRAINT fk_test_patient FOREIGN KEY (Patient_id) REFERENCES PATIENT(Patient_id)
);

-- From: VALID (P_id, valid) - Extracted from Bills during normalization
CREATE TABLE VALID (
    P_id INT PRIMARY KEY,
    Valid VARCHAR(50),
    CONSTRAINT fk_valid_patient FOREIGN KEY (P_id) REFERENCES PATIENT(Patient_id)
);

-- From: BILLS (B_id, P_id, Amount, I_amount)
-- EER Constraint: Weak entity on Patient ('Pays')
CREATE TABLE BILLS (
    B_id INT,
    P_id INT,
    Amount DECIMAL(10, 2),
    I_amount DECIMAL(10, 2),
    PRIMARY KEY (B_id, P_id),
    CONSTRAINT fk_bills_valid FOREIGN KEY (P_id) REFERENCES VALID(P_id)
);

-- ==============================================================================
-- 5. ASSOCIATIONS & M:N RELATIONSHIPS (Satisfying the EER Diagram links)
-- ==============================================================================

-- From: MEDICAL_RECORDS (R_id, P_id, Purchase_date)
CREATE TABLE MEDICAL_RECORDS (
    R_id INT,
    P_id INT,
    Purchase_date DATE,
    PRIMARY KEY (R_id, P_id),
    CONSTRAINT fk_medrec_patient FOREIGN KEY (P_id) REFERENCES PATIENT(Patient_id)
);

-- From: RECORD_HANDLER (P_id, Purchase_date, Emp_id)
-- EER Constraint: M:N 'Maintains' relationship managed by Pharmacist
CREATE TABLE RECORD_HANDLER (
    P_id INT,
    Purchase_date DATE,
    Emp_id INT,
    PRIMARY KEY (P_id, Purchase_date, Emp_id),
    CONSTRAINT fk_rechandler_patient FOREIGN KEY (P_id) REFERENCES PATIENT(Patient_id),
    CONSTRAINT fk_rechandler_pharmacist FOREIGN KEY (Emp_id) REFERENCES PHARMACIST(Emp_id)
);

-- From: APPOINTMENT (Id, date, P_id)
CREATE TABLE APPOINTMENT (
    Id INT PRIMARY KEY,
    App_date DATE,
    P_id INT,
    CONSTRAINT fk_appointment_patient FOREIGN KEY (P_id) REFERENCES PATIENT(Patient_id)
);

-- From: Appointment_B (P_id, D_id)
-- EER Constraint: M:N 'Consulted_by' bridging Patient/Appointment to Doctor
CREATE TABLE Appointment_B (
    P_id INT,
    D_id INT,
    PRIMARY KEY (P_id, D_id),
    CONSTRAINT fk_appB_patient FOREIGN KEY (P_id) REFERENCES PATIENT(Patient_id),
    CONSTRAINT fk_appB_doctor FOREIGN KEY (D_id) REFERENCES DOCTOR(Emp_id)
);

-- ADDED TO FULFILL EER DIAGRAM: Nurse 'Handles' Rooms (M:N)
-- This was drawn in the EER diagram but missed in the handwritten normalized list.
CREATE TABLE NURSE_HANDLES_ROOMS (
    Emp_id INT,
    Room_no INT,
    PRIMARY KEY (Emp_id, Room_no),
    CONSTRAINT fk_nhr_nurse FOREIGN KEY (Emp_id) REFERENCES NURSE(Emp_id),
    CONSTRAINT fk_nhr_room FOREIGN KEY (Room_no) REFERENCES ROOMS(Room_no)
);

1. Basic Filtering (WHERE clause)
This query retrieves the room numbers and capacities of all rooms that are specifically designated as 'General' wards.

SELECT 
    Room_no, 
    Capacity 
FROM ROOMS 
WHERE Type = 'General';

==================================================================

2. Finding Missing Data (IS NULL)
This query identifies all patients who are currently admitted to the hospital (meaning their discharge date has not yet been recorded).


SELECT 
    Patient_id, 
    F_name, 
    L_name, 
    In_date 
FROM PATIENT 
WHERE Out_date IS NULL;

==================================================================

3. Simple Aggregation (GROUP BY)
This query counts the number of nurses assigned to each type of shift (e.g., Morning, Night).


SELECT 
    Shift_type, 
    COUNT(Emp_id) AS Total_Nurses 
FROM NURSE 
GROUP BY Shift_type;

==================================================================

4. Sorting Data (ORDER BY)
This query retrieves a list of all employees over the age of 40, sorted from oldest to youngest.


SELECT 
    Emp_id, 
    F_name, 
    L_name, 
    Age 
FROM EMPLOYEE 
WHERE Age > 40 
ORDER BY Age DESC;

==================================================================

5. Simple 2-Table JOIN
This query fetches the names and clearance levels of all pharmacists by linking the base EMPLOYEE table with the PHARMACIST subclass table.


SELECT 
    e.F_name, 
    e.L_name, 
    p.Clearance_level 
FROM EMPLOYEE e
JOIN PHARMACIST p ON e.Emp_id = p.Emp_id;

==================================================================

6. Multi-Table JOIN (4 Tables) - 'Consulted_by' Relationship
This query lists every patient along with the doctor they consulted and the doctor's
specialization, by traversing PATIENT -> APPOINTMENT_B -> DOCTOR -> EMPLOYEE.


SELECT 
    p.Patient_id,
    p.F_name || ' ' || p.L_name AS Patient_Name,
    e.F_name || ' ' || e.L_name AS Doctor_Name,
    d.Specialization
FROM PATIENT p
JOIN Appointment_B ab ON p.Patient_id = ab.P_id
JOIN DOCTOR d ON ab.D_id = d.Emp_id
JOIN EMPLOYEE e ON d.Emp_id = e.Emp_id
ORDER BY p.Patient_id;

==================================================================

7. LEFT JOIN (Outer Join)
This query lists all patients with their bills, but also includes patients who do
not have any bill yet (e.g., newly admitted patients). Their bill columns are NULL.


SELECT 
    p.Patient_id,
    p.F_name,
    p.L_name,
    b.B_id,
    b.Amount,
    b.I_amount
FROM PATIENT p
LEFT JOIN BILLS b ON p.Patient_id = b.P_id
ORDER BY p.Patient_id;

==================================================================

8. Aggregation with HAVING
This query computes the total billed amount and insurance amount per patient and
returns only those patients whose total bill exceeds 1500, sorted high to low.


SELECT 
    p.Patient_id,
    p.F_name,
    p.L_name,
    SUM(b.Amount) AS Total_Billed,
    SUM(b.I_amount) AS Total_Insurance
FROM PATIENT p
JOIN BILLS b ON p.Patient_id = b.P_id
GROUP BY p.Patient_id, p.F_name, p.L_name
HAVING SUM(b.Amount) > 1500
ORDER BY Total_Billed DESC;

==================================================================

9. Recursive JOIN (Unary Relationship) - 'Supervision'
This query resolves the recursive Supervision relationship on DOCTOR by joining
the table to itself, showing each doctor alongside their supervisor's name.


SELECT 
    d.Emp_id AS Doctor_ID,
    doc.F_name || ' ' || doc.L_name AS Doctor_Name,
    s.Emp_id AS Supervisor_ID,
    sup.F_name || ' ' || sup.L_name AS Supervisor_Name
FROM DOCTOR d
JOIN EMPLOYEE doc ON d.Emp_id = doc.Emp_id
LEFT JOIN DOCTOR s ON d.Supervisor_id = s.Emp_id
LEFT JOIN EMPLOYEE sup ON s.Emp_id = sup.Emp_id
ORDER BY d.Emp_id;

==================================================================

10. Subquery with IN (Weak Entity Check)
This query finds all patients who have at least one dependent registered in the
weak entity PERSONS ('Dependent_On' relationship).


SELECT 
    Patient_id,
    F_name,
    L_name,
    Gender
FROM PATIENT
WHERE Patient_id IN (SELECT Patient_id FROM PERSONS);

==================================================================

11. Subquery with Aggregate Function
This query returns the bills whose amount is strictly greater than the average
bill amount across the whole hospital.


SELECT 
    B_id,
    P_id,
    Amount,
    I_amount
FROM BILLS
WHERE Amount > (SELECT AVG(Amount) FROM BILLS);

==================================================================

12. Five-Table JOIN - Nurse 'Handles' Rooms (M:N)
This query shows which nurse handles which room, including the nurse's shift,
room type and availability, by joining the M:N table with both subclass and
base tables (NURSE_HANDLES_ROOMS -> NURSE -> EMPLOYEE -> ROOMS -> ROOMS_B).


SELECT 
    e.Emp_id,
    e.F_name || ' ' || e.L_name AS Nurse_Name,
    n.Shift_type,
    r.Room_no,
    r.Type AS Room_Type,
    rb.Availability
FROM NURSE_HANDLES_ROOMS nhr
JOIN NURSE n ON nhr.Emp_id = n.Emp_id
JOIN EMPLOYEE e ON n.Emp_id = e.Emp_id
JOIN ROOMS r ON nhr.Room_no = r.Room_no
JOIN ROOMS_B rb ON r.Capacity = rb.Capacity
ORDER BY e.Emp_id, r.Room_no;

==================================================================

13. EXISTS Correlated Subquery
This query finds patients who are currently admitted (no discharge date) AND
already have at least one test report recorded.


SELECT 
    p.Patient_id,
    p.F_name,
    p.L_name
FROM PATIENT p
WHERE p.Out_date IS NULL
  AND EXISTS (
        SELECT 1 
        FROM TEST_REPORT tr 
        WHERE tr.Patient_id = p.Patient_id
  );

==================================================================

14. Composite-Key JOIN - Pharmacist 'Maintains' Medical Records
This query joins MEDICAL_RECORDS with RECORD_HANDLER on the composite key
(P_id, Purchase_date) to show who handled each record, plus the pharmacist's
clearance level from the subclass table.


SELECT 
    mr.R_id,
    p.F_name || ' ' || p.L_name AS Patient_Name,
    mr.Purchase_date,
    e.F_name || ' ' || e.L_name AS Pharmacist_Name,
    ph.Clearance_level
FROM MEDICAL_RECORDS mr
JOIN RECORD_HANDLER rh 
     ON mr.P_id = rh.P_id AND mr.Purchase_date = rh.Purchase_date
JOIN PHARMACIST ph ON rh.Emp_id = ph.Emp_id
JOIN EMPLOYEE e ON ph.Emp_id = e.Emp_id
JOIN PATIENT p ON mr.P_id = p.Patient_id
ORDER BY mr.Purchase_date;

==================================================================

15. CASE Expression (Derived Column)
This query computes what percentage of each bill was covered by insurance,
using a CASE expression to guard against division by zero.


SELECT 
    B_id,
    P_id,
    Amount,
    I_amount,
    CASE 
        WHEN Amount = 0 THEN 0
        ELSE ROUND((I_amount / Amount) * 100, 2)
    END AS Insurance_Coverage_Pct
FROM BILLS;

==================================================================

16. Date Arithmetic (Length of Stay)
This query calculates how many days each discharged patient stayed in the
hospital. (For MySQL, replace the expression with DATEDIFF(Out_date, In_date).)


SELECT 
    Patient_id,
    F_name,
    L_name,
    In_date,
    Out_date,
    (Out_date - In_date) AS Stay_Days
FROM PATIENT
WHERE Out_date IS NOT NULL;

==================================================================

17. CREATE VIEW + Querying a View
This query creates a reusable view of all currently admitted patients and then
queries it. Views abstract the base logic and can be joined like real tables.


CREATE VIEW CURRENT_IN_PATIENTS AS
SELECT 
    Patient_id,
    F_name || ' ' || L_name AS Patient_Name,
    Gender,
    Phone,
    In_date
FROM PATIENT
WHERE Out_date IS NULL;

-- Query the newly created view
SELECT * 
FROM CURRENT_IN_PATIENTS;
